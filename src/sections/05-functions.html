المرحلة الخامسة

الدوال، الـ Scope، ومدة حياة المتغيرات

أول حاجة بتعملها في أي sample: تقسّم الكود لدوال وتسميها. لما تفهم الدالة بتاخد إيه وبترجع إيه وبتلمس أنهي ذاكرة، تقدر تسمّيها (decrypt_config، send_beacon…) وتكمل. المرحلة دي بتعلمك الدوال من جوه C ومن جوه الـ CPU.

التعريف، الـ Prototype، والاستدعاء functions

#include #include /* Prototype (declaration): tells the compiler the signature */ uint32_t checksum(const unsigned char *data, size_t len); static void banner(void); int main(void) { const unsigned char payload[] = { 0x10, 0x20, 0x30, 0xFF }; banner(); printf("checksum = 0x%08X\n", checksum(payload, sizeof payload)); return 0; } /* Definition: the actual body */ uint32_t checksum(const unsigned char *data, size_t len) { uint32_t sum = 0; for (size_t i = 0; i < len; i++) sum = (sum << 1) ^ data[i]; return sum; } static void banner(void) { /* void = returns nothing */ puts("== integrity check =="); }

Pass by Value: الدالة بتاخد نسخة copies

في C، كل الـ arguments بتتبعت بالقيمة — الدالة بتاخد نسخة. عشان دالة تعدّل متغير عند اللي ناداها، لازم تبعتلها عنوانه (pointer) — وده أول سبب عملي للـ pointers.

#include static void try_reset(int x) { x = 0; } /* modifies its copy only */ static void real_reset(int *x) { *x = 0; } /* modifies the original */ static void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; } int main(void) { int v = 99; try_reset(v); printf("after try_reset : %d\n", v); real_reset(&v); printf("after real_reset: %d\n", v); int p = 1, q = 2; swap(&p, &q); printf("after swap: p=%d q=%d\n", p, q); return 0; }

لاحظ لما تشغّل: الـ compiler بيطلع warning إن x في try_reset "set but not used" — هو نفسه شايف إن التعديل ده ملوش أي أثر برا الدالة.

الدالة من جوه الـ CPU registers

على Linux x86-64 (System V ABI) أول 6 arguments بتتبعت في registers بالترتيب: rdi, rsi, rdx, rcx, r8, r9 (والنسخ الـ 32-bit: edi, esi, edx, ecx…)، والقيمة الراجعة في rax/eax. ده ناتج gcc الحقيقي:

int add3(int a, int b, int c) { int sum = a + b + c; return sum; } int caller(void) { return add3(1, 2, 3); } add3: push rbp mov rbp, rsp mov DWORD PTR [rbp-20], edi ; a (1st arg) mov DWORD PTR [rbp-24], esi ; b (2nd arg) mov DWORD PTR [rbp-28], edx ; c (3rd arg) mov edx, DWORD PTR [rbp-20] mov eax, DWORD PTR [rbp-24] add edx, eax ; a + b mov eax, DWORD PTR [rbp-28] add eax, edx ; + c mov DWORD PTR [rbp-4], eax ; sum mov eax, DWORD PTR [rbp-4] ; return in eax pop rbp ret caller: push rbp mov rbp, rsp mov edx, 3 ; args are loaded mov esi, 2 ; right-to-left mov edi, 1 call add3 pop rbp ret

وفي -O2؟ add3 بقت add edi, esi + lea eax, [rdi+rdx]، وcaller بقت mov eax, 6 بس — الـ compiler حسب النتيجة وقت الـ compile وشال الـ call خالص (inlining + constant folding). عشان كده في الـ binaries المعمول لها optimization، دوال صغيرة كتير بتختفي وكودها بيندمج جوه اللي بيناديها.

🔬
الـ static local في الـ decompiler = global

المتغير الـ static جوه دالة بيتخزن في .data زي أي global — gcc بيسميه حاجة زي calls.0 وبيوصله بـ [rip+offset]. في Ghidra هيظهر كـ DAT_00104010. لما تشوف دالة بتقرا وتكتب في global بتاعها لوحدها، فكّر: counter، flag "initialized"، أو cache — زي الـ malware اللي بيحل الـ APIs مرة واحدة ويخزنها.

الـ Recursion والـ Stack recursion

الدالة اللي بتنادي نفسها. كل استدعاء بياخد stack frame جديد، فلو مفيش شرط توقف (base case) أو العمق كبير جدًا = Stack Overflow والبرنامج بيقع.

#include static unsigned long long factorial(unsigned n) { if (n <= 1) return 1; /* base case */ return n * factorial(n - 1); /* recursive case */ } static unsigned gcd(unsigned a, unsigned b) { return b == 0 ? a : gcd(b, a % b); /* tail recursion */ } static void depth(int level) { if (level > 3) return; printf("%*sdepth %d\n", level * 2, "", level); depth(level + 1); printf("%*sunwind %d\n", level * 2, "", level); } int main(void) { printf("10! = %llu\n", factorial(10)); printf("gcd(1071, 462) = %u\n", gcd(1071, 462)); depth(1); return 0; }
💡
جرّب ASM على gcd بـ -O2

الـ tail recursion (لما الاستدعاء هو آخر حاجة) gcc بيحولها لـ loop عادية — مفيش call خالص. عشان كده مش كل دالة recursive في الكود تبان recursive في الـ binary.

Variadic Functions: زي printf stdarg.h

#include #include /* A logger like the ones you will find inside malware debug builds */ static void log_msg(const char *level, const char *fmt, ...) { va_list args; va_start(args, fmt); printf("[%s] ", level); vprintf(fmt, args); /* forward the variable arguments */ printf("\n"); va_end(args); } static int sum_n(int count, ...) { va_list ap; va_start(ap, count); int total = 0; for (int i = 0; i < count; i++) total += va_arg(ap, int); va_end(ap); return total; } int main(void) { log_msg("INFO", "connected to %s:%d", "10.0.0.5", 443); log_msg("WARN", "retry %d of %d", 2, 5); printf("sum = %d\n", sum_n(4, 10, 20, 30, 40)); return 0; }

main و الـ Command Line argc · argv

#include int main(int argc, char *argv[]) { printf("argc = %d\n", argc); for (int i = 0; i < argc; i++) printf("argv[%d] = %s\n", i, i == 0 ? "" : argv[i]); if (argc < 2) { fprintf(stderr, "usage: %s \n", "tool"); return 1; } return 0; }

شغّل المثال: هيرجع 1 لأن مفيش arguments. argv[0] = مسار البرنامج، وargv[argc] = NULL دايمًا. والـ malware كتير بيتحكم في سلوكه بالـ arguments (مثلًا -install أو -svc) — فلو الـ sample مبيعملش حاجة في الـ sandbox، شوف هل بيدور على argument معين في main.
