المرحلة العشرون

الـ Sockets: إزاي البرنامج بيكلم الشبكة (وإزاي الـ C2 بيشتغل)

كل backdoor وكل RAT وكل bot لازم يتكلم مع الـ C2 (Command & Control). تحت أي مكتبة HTTP، فيه sockets. لما تعرف الـ API ده كويس، هتعرف تطلع الـ IP والـ port من الـ disassembly، وتفهم الـ protocol اللي الـ malware بيتكلم بيه، وتكتب أداة تفك الـ traffic اللي في الـ PCAP.

الصورة الكبيرة: Client و Server TCP · UDP

الـ socket = طرف اتصال، والـ kernel بيرجعهولك كـ file descriptor (على Linux) أو SOCKET (على Windows). TCP = اتصال موثوق ومرتب (stream of bytes)، وUDP = رسائل منفصلة (datagrams) من غير ضمان وصول. الـ malware غالبًا TCP (أو HTTP/HTTPS فوقه)، وUDP بيظهر في DNS tunneling وفي هجمات DDoS بتاعة الـ bots.

Network Byte Order و sockaddr_in htons · inet_pton

الشبكة بتستخدم big-endian ("network byte order")، وx86 little-endian (المرحلة 02). عشان كده الـ port والـ IP لازم يتحولوا قبل ما يتحطوا في الـ struct: htons (host to network short) للـ port، وhtonl للـ 32-bit، وntohs/ntohl للعكس. والعنوان نفسه بيتحط في struct sockaddr_in:

#define _POSIX_C_SOURCE 200809L #include #include #include #include static void dump(const char *label, const void *p, size_t n) { const unsigned char *b = p; printf("%-18s", label); for (size_t i = 0; i < n; i++) printf("%02X ", b[i]); printf("\n"); } int main(void) { struct sockaddr_in a; memset(&a, 0, sizeof a); a.sin_family = AF_INET; /* 2, host order */ a.sin_port = htons(443); /* 0x01BB -> bytes 01 BB */ inet_pton(AF_INET, "192.168.1.10", &a.sin_addr); printf("sizeof(sockaddr_in) = %zu\n", sizeof a); dump("family (2 bytes):", &a.sin_family, 2); dump("port (2 bytes):", &a.sin_port, 2); dump("addr (4 bytes):", &a.sin_addr, 4); printf("port read as a little-endian WORD = 0x%04X\n", a.sin_port); printf("addr read as a little-endian DWORD = 0x%08X\n", a.sin_addr.s_addr); printf("back to host order: port %u\n", ntohs(a.sin_port)); char text[INET_ADDRSTRLEN]; inet_ntop(AF_INET, &a.sin_addr, text, sizeof text); printf("back to text: %s\n", text); return 0; }

طلّع الـ C2 من الأرقام

كتير من الـ samples مش بتخزن الـ IP كـ string. هتلاقي في الـ disassembly: mov word ptr [rsp+20h], 2 (الـ family) ← mov word ptr [rsp+22h], 0BB01h ← mov dword ptr [rsp+24h], 0A01A8C0h. فكّها: 0xBB01 مقلوب = 0x01BB = 443. و0x0A01A8C0 = الـ bytes C0 A8 01 0A = 192.168.1.10. وفي Ghidra غيّر نوع المتغير لـ sockaddr_in والحقول هتبان لوحدها. ولو لقيت call htons مع ثابت زي 0x115C = port 4444 (الـ default بتاع Metasploit).

Server و Client حقيقيين (على loopback) TCP echo

المثال ده بيعمل server على 127.0.0.1، ويسيب الـ kernel يختار port فاضي (port 0)، وبعدين fork: الـ child بيبقى client بيتصل ويبعت رسالة، والـ parent (الـ server) بيرد عليه.

#define _POSIX_C_SOURCE 200809L #include #include #include #include #include #include #include #include int main(void) { /* ---------- server setup ---------- */ int srv = socket(AF_INET, SOCK_STREAM, 0); if (srv < 0) { perror("socket"); return 1; } struct sockaddr_in addr = { 0 }; addr.sin_family = AF_INET; addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK); /* 127.0.0.1 */ addr.sin_port = 0; /* let the kernel pick */ if (bind(srv, (struct sockaddr *)&addr, sizeof addr) != 0) { perror("bind"); return 1; } listen(srv, 1); socklen_t len = sizeof addr; getsockname(srv, (struct sockaddr *)&addr, &len); /* which port did we get? */ fflush(stdout); pid_t pid = fork(); if (pid == 0) { /* ---------- client ---------- */ close(srv); int c = socket(AF_INET, SOCK_STREAM, 0); if (connect(c, (struct sockaddr *)&addr, sizeof addr) != 0) { perror("connect"); _exit(1); } const char *msg = "beacon id=42 os=linux"; send(c, msg, strlen(msg), 0); char reply[64] = { 0 }; ssize_t n = recv(c, reply, sizeof reply - 1, 0); printf("[client] got %zd bytes: %s\n", n, reply); fflush(stdout); close(c); _exit(0); } /* ---------- server: serve one client ---------- */ int cli = accept(srv, NULL, NULL); char buf[64] = { 0 }; ssize_t n = recv(cli, buf, sizeof buf - 1, 0); for (ssize_t i = 0; i < n; i++) buf[i] = (char)toupper((unsigned char)buf[i]); send(cli, buf, (size_t)n, 0); /* echo it back, uppercased */ close(cli); close(srv); waitpid(pid, NULL, 0); printf("[server] handled 1 client, %zd bytes\n", n); return 0; }
⚠️
recv مش بيرجع "رسالة"

TCP = stream. لو الطرف التاني بعت 1000 byte، ممكن recv يرجع 300 وبعدين 700، أو يجمع رسالتين في مرة. وsend ممكن يبعت جزء بس. عشان كده كل protocol حقيقي (وكل C2) لازم يحدد حدود الرسالة: إما delimiter (زي \r\n في HTTP) أو length prefix (الطول في الأول). وكل كود محترم فيه loop زي ده:

#define _POSIX_C_SOURCE 200809L #include #include #include #include #include /* Keep calling send() until every byte is out (or an error happens). */ int send_all(int fd, const void *data, size_t len) { const char *p = data; while (len > 0) { ssize_t n = send(fd, p, len, 0); if (n <= 0) return -1; /* error or connection closed */ p += n; len -= (size_t)n; } return 0; } /* Read EXACTLY len bytes (needed for length-prefixed protocols). */ int recv_exact(int fd, void *data, size_t len) { char *p = data; while (len > 0) { ssize_t n = recv(fd, p, len, 0); if (n <= 0) return -1; p += n; len -= (size_t)n; } return 0; } int main(void) { int sv[2]; if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) != 0) { perror("socketpair"); return 1; } send_all(sv[0], "0123456789", 10); char buf[11] = { 0 }; if (recv_exact(sv[1], buf, 4) == 0) printf("first 4 bytes : %s\n", buf); if (recv_exact(sv[1], buf, 6) == 0) printf("next 6 bytes : %.6s\n", buf); close(sv[0]); close(sv[1]); return 0; }

بروتوكول C2 بسيط: Length-Prefixed Messages framing

أغلب الـ C2 protocols المخصصة شكلها: header ثابت (نوع الأمر + الطول) وبعدين payload، وغالبًا مشفّر بـ XOR أو RC4. المثال ده بيبني رسالة ويفكها، فوق socketpair (اتنين sockets متوصلين ببعض جوه نفس البرنامج):

#define _POSIX_C_SOURCE 200809L #include #include #include #include #include /* wire format: [type:1][length:2, big-endian][payload:length] */ static size_t build(uint8_t *out, uint8_t type, const void *payload, uint16_t len) { out[0] = type; out[1] = (uint8_t)(len >> 8); /* big-endian length */ out[2] = (uint8_t)(len & 0xFF); memcpy(out + 3, payload, len); return 3u + len; } int main(void) { int sv[2]; if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) != 0) { perror("socketpair"); return 1; } /* "C2 side" sends two commands back to back */ uint8_t wire[128]; size_t n = build(wire, 0x10, "whoami", 6); n += build(wire + n, 0x20, "\x00\x3C", 2); /* sleep 60 (big-endian) */ if (write(sv[0], wire, n) != (ssize_t)n) return 1; close(sv[0]); /* "implant side" parses the stream message by message */ uint8_t hdr[3], payload[256]; while (read(sv[1], hdr, 3) == 3) { uint16_t len = (uint16_t)((hdr[1] << 8) | hdr[2]); if (read(sv[1], payload, len) != len) { puts("truncated"); break; } if (hdr[0] == 0x10) printf("cmd 0x10 EXEC len=%u arg=\"%.*s\"\n", len, (int)len, (const char *)payload); else if (hdr[0] == 0x20) printf("cmd 0x20 SLEEP len=%u seconds=%u\n", len, (payload[0] << 8) | payload[1]); } close(sv[1]); return 0; }
🔬
إزاي تفك protocol مجهول

(1) breakpoint بعد recv (أو InternetReadFile) واعمل dump للـ buffer. (2) دوّر على الـ parser: أول bytes بتتقرا وبتتقارن بثوابت (cmp al, 10h) = الـ command IDs، وغالبًا بعدها switch/jump table (المرحلة 04) أو dispatch table (المرحلة 08). (3) القيمة اللي بتتعمل لها bswap أو shl 8 | … = طول big-endian. (4) اكتب parser بـ Python أو C يفك الـ PCAP كله. ده بالظبط اللي بيتعمل في الـ threat reports.

UDP: رسائل منفصلة sendto · recvfrom

الكود الحديث مش بيحط IP ثابت: بيحل اسم domain بـ getaddrinfo (الطريقة الصح، بتدعم IPv4 وIPv6). سيرفر التشغيل هنا معندوش DNS، فجرّب المثال ده على جهازك:

DGA و DNS في الـ malware

بعض العائلات بتولّد مئات الـ domains كل يوم بخوارزمية (DGA = Domain Generation Algorithm) من التاريخ أو seed، وبتجرب تحلهم واحد واحد. المحلل بيفك الخوارزمية من الـ disassembly ويعيد كتابتها بـ C أو Python عشان يعرف الـ domains مقدمًا ويعملها block أو sinkhole. علامتها: loop فيه LCG أو hash على أرقام اليوم/الشهر/السنة، وبيبني string من 'a' + (x % 26)، وبعدها getaddrinfo/DnsQuery_A.

أكتر من اتصال في نفس الوقت poll · select

recv بيقف (blocking) لحد ما تيجي بيانات. عشان server يخدم كذا client، أو implant يستنى الـ C2 والـ stdin بتاع shell في نفس الوقت، فيه 3 حلول: thread لكل اتصال (المرحلة 18)، أو poll/select (استنى على كذا descriptor مرة واحدة)، أو non-blocking sockets (O_NONBLOCK / ioctlsocket(FIONBIO) على Windows). لو شفت select في loop مع socket وpipe، ده غالبًا shell تفاعلي بينقل البيانات في الاتجاهين.

شكل الـ Beacon

الـ implant النموذجي: loop لا نهائي فيه (1) جمع معلومات (hostname، user، OS) ← (2) request للـ C2 ← (3) parse للرد وتنفيذ الأمر ← (4) Sleep(interval ± jitter). الـ jitter (رقم عشوائي بيتضاف على وقت النوم) عشان الـ traffic ميبقاش منتظم والـ IDS ميكتشفهوش. في الـ sandbox ممكن متشوفش أي traffic لأن الـ sleep طويل، وده سبب إن أدوات زي FakeNet-NG وتعديل قيمة الـ sleep في الـ debugger مهمين.
