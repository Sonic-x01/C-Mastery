المرحلة الرابعة

الشروط والحلقات — وشكلها في الـ Disassembler

في الـ Assembly مفيش if ولا for ولا while. فيه بس مقارنة (cmp/test) وقفزة (jmp/jcc). كل الـ control flow اللي هتكتبه بيتحوّل لكده — وفي IDA/Ghidra هتشوفه كـ graph من blocks وأسهم. المرحلة دي بتعلمك تقرا الـ graph ده.

if / else if / else branches

#include static const char *classify(int score) { if (score < 0 || score > 100) return "invalid"; else if (score >= 85) return "malicious"; else if (score >= 50) return "suspicious"; else return "clean"; } int main(void) { int scores[] = { 97, 63, 12, -5 }; for (int i = 0; i < 4; i++) printf("score %3d -> %s\n", scores[i], classify(scores[i])); int x = 0; if (x) /* any non-zero value is "true" */ puts("x is true"); else puts("x is false (0)"); return 0; }
⚠️
الـ true في C = أي رقم غير الصفر

مفيش boolean حقيقي قبل C99 (_Bool / bool من <stdbool.h>). الشرط بيتحقق لو القيمة ≠ 0. وغلطة if (x = 5) (تخصيص بدل مقارنة) دايمًا true — -Wall بيحذرك منها.

الـ if في الـ Assembly: الشرط بيتعكس -O0

الـ compiler بيكتب الكود بحيث يقفز فوق جسم الـ if لو الشرط اتكسر. فـ if (age >= 18) بيتحول لـ "لو age <= 17 اقفز للـ else". ده ناتج gcc الحقيقي:

int check_age(int age) { if (age >= 18) return 1; return 0; } check_age: push rbp mov rbp, rsp mov DWORD PTR [rbp-4], edi ; age -> local cmp DWORD PTR [rbp-4], 17 ; compare age with 17 jle .L2 ; age <= 17 ? skip the "then" mov eax, 1 ; then: return 1 jmp .L3 .L2: mov eax, 0 ; else: return 0 .L3: pop rbp ret

لاحظ: >= 18 اتكتبت cmp 17 + jle (الشرط المعكوس). وفي -O2 الـ branch كله بيختفي: cmp edi, 17 ← setg al (حط 1 لو أكبر). جرّب زرار ASM على أي مثال وبدّل الـ optimization.

switch و Jump Tables switch

#include static const char *cmd_name(int cmd) { switch (cmd) { case 1: return "BEACON"; case 2: return "SHELL"; case 3: return "UPLOAD"; case 4: return "DOWNLOAD"; case 5: return "SLEEP"; default: return "UNKNOWN"; } } int main(void) { for (int c = 0; c <= 6; c++) printf("cmd %d -> %s\n", c, cmd_name(c)); /* Fall-through: no break means execution continues into the next case */ int level = 2; switch (level) { case 3: puts(" enable keylog"); /* fall through */ case 2: puts(" enable screenshots"); /* fall through */ case 1: puts(" enable beacon"); break; default: puts(" nothing"); } return 0; }

لما الـ cases تبقى أرقام متقاربة (dense)، الـ compiler بيعمل Jump Table: array من العناوين، والـ case رقم n بيقفز للعنوان رقم n — مقارنة واحدة بدل سلسلة طويلة. ده ناتج gcc -O2 حقيقي:

void f1(void); void f2(void); void f3(void); void f4(void); void f5(void); void run(int cmd) { switch (cmd) { case 1: f1(); break; case 2: f2(); break; case 3: f3(); break; case 4: f4(); break; case 5: f5(); break; } } run: cmp edi, 5 ; UNSIGNED compare: also rejects negatives ja .L1 ; cmd > 5 (or < 0) -> default mov edi, edi ; zero-extend index to 64-bit jmp [QWORD PTR .L4[0+rdi*8]] ; jump through the table .L4: ; the jump table (in .rodata) .quad .L1 ; case 0 -> default .quad .L8 ; case 1 .quad .L7 ; case 2 .quad .L6 ; case 3 .quad .L5 ; case 4 .quad .L3 ; case 5 .L3: jmp f5 ; tail calls .L8: jmp f1 .L7: jmp f2 .L6: jmp f3 .L5: jmp f4 .L1: ret
🔬
الـ Jump Table = الـ Command Dispatcher في الـ RATs

الـ backdoors بتستقبل رقم أمر من الـ C2 وتعمل switch عليه. لما تلاقي cmp reg, N + ja + jmp [table+reg*8] — لقيت قلب الـ malware: كل entry في الجدول = قدرة (capability). المحلل بيمشي على كل case ويوثّق هي بتعمل إيه. Ghidra وIDA غالبًا بيتعرفوا على الجدول وبيرجعوه switch.

💡
أشكال تانية للـ switch

لو الـ cases بترجع قيم بس، gcc ممكن يحولها لـ lookup table للبيانات مش للعناوين (هتشوف array اسمه CSWTCH). ولو الـ cases متفرقة (1، 100، 5000) بيعمل سلسلة مقارنات أو binary search من الـ cmp.

الحلقات: while / do-while / for loops

#include int main(void) { /* while: test first, may run 0 times */ int n = 3; while (n > 0) { printf("while n=%d\n", n); n--; } /* do-while: body first, runs at least once */ int tries = 0; do { tries++; printf("attempt %d\n", tries); } while (tries < 2); /* for: init; condition; step */ for (int i = 0; i < 10; i += 3) printf("for i=%d\n", i); /* break / continue */ for (int i = 1; i <= 10; i++) { if (i % 2 == 0) continue; /* skip evens */ if (i > 7) break; /* stop early */ printf("odd %d\n", i); } /* nested loops with a flag */ int found = 0; for (int r = 0; r < 3 && !found; r++) for (int c = 0; c < 3; c++) if (r * 3 + c == 7) { printf("found at r=%d c=%d\n", r, c); found = 1; break; } return 0; }

شكل الـ loop في الـ Assembly back-edge

int sum_to(int n) { int total = 0; for (int i = 1; i <= n; i++) total += i; return total; } sum_to: push rbp mov rbp, rsp mov DWORD PTR [rbp-20], edi ; n mov DWORD PTR [rbp-4], 0 ; total = 0 mov DWORD PTR [rbp-8], 1 ; i = 1 jmp .L5 ; jump to the condition first .L6: ; loop body mov eax, DWORD PTR [rbp-8] add DWORD PTR [rbp-4], eax ; total += i add DWORD PTR [rbp-8], 1 ; i++ .L5: ; condition mov eax, DWORD PTR [rbp-8] cmp eax, DWORD PTR [rbp-20] jle .L6 ; i <= n ? back to body (BACK-EDGE) mov eax, DWORD PTR [rbp-4] ; return total pop rbp ret
🔬
إزاي تعرف الـ loop في أي disassembly

دوّر على قفزة لورا (back-edge): jump لعنوان أصغر من عنوانها. في IDA graph بيظهر كسهم راجع لفوق. بعدها حدد: (1) الـ counter (المتغير اللي بيزيد بـ add ..., 1)، (2) شرط الخروج (الـ cmp قبل الـ jump)، (3) الجسم. في -O2 الـ compiler ممكن يعمل loop unrolling (يكرر الجسم مرتين أو أكتر في اللفة الواحدة) أو يستبدل الـ loop بمعادلة — جرّب ASM على المثال وشوف.

goto: ممنوع غالبًا… إلا هنا cleanup

الـ goto بيقفز لـ label في نفس الدالة. استخدامه المقبول الوحيد في C الحديث: تنظيف الموارد عند الأخطاء (أسلوب Linux kernel). ومهم تعرفه لأن الـ decompilers بتطلّع goto كتير لما الـ control flow يكون معقد أو متعمله obfuscation.

#include #include static int process(int fail_at) { int rc = -1; char *a = NULL, *b = NULL; a = malloc(64); if (!a || fail_at == 1) goto cleanup; b = malloc(64); if (!b || fail_at == 2) goto cleanup; puts(" work done"); rc = 0; cleanup: /* single exit point: free everything */ free(b); /* free(NULL) is safe */ free(a); return rc; } int main(void) { printf("run ok -> rc=%d\n", process(0)); printf("fail @1 -> rc=%d\n", process(1)); printf("fail @2 -> rc=%d\n", process(2)); return 0; }
