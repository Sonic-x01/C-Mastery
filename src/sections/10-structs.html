المرحلة العاشرة

Structs، Unions، Enums — وإعادة بناء الـ Structs من الـ Assembly

الـ struct = مجموعة fields ورا بعض في الذاكرة. في الـ binary الأسماء بتختفي وبيفضل بس offsets: [rdi+4]، [rdi+24]. أهم مهارة في الـ RE بعد قراية الـ control flow هي إنك تعيد بناء الـ struct من الـ offsets دي. المرحلة دي بتعلمك الـ layout بالـ byte.

التعريف والاستخدام struct

#include #include #include struct process_info { uint32_t pid; uint32_t ppid; char name[16]; int is_elevated; }; typedef struct process_info ProcInfo; /* shorter name */ static void print_proc(const ProcInfo *p) { /* pass by pointer: no copy */ printf("%-13s pid=%-5u ppid=%-5u %s\n", p->name, p->pid, p->ppid, p->is_elevated ? "[ELEVATED]" : ""); } int main(void) { ProcInfo a = { 4321, 1200, "malware.exe", 1 }; ProcInfo b = { .pid = 1200, .name = "explorer.exe" }; /* designated init */ b.ppid = 900; print_proc(&a); print_proc(&b); ProcInfo copy = a; /* struct assignment copies all bytes */ strcpy(copy.name, "copy.exe"); printf("original still: %s, sizeof = %zu\n", a.name, sizeof(ProcInfo)); return 0; }

s.field لما يكون عندك الـ struct نفسه، وp->field (اختصار (*p).field) لما يكون عندك pointer له. وفي الـ assembly الاتنين بيبقوا العنوان + offset.

Padding و Alignment layout

الـ CPU بيحب يقرا كل نوع من عنوان مضاعف لحجمه (int من عنوان يقبل القسمة على 4، uint64_t على 8). فالـ compiler بيحط padding (bytes فاضية) بين الـ fields، وفي آخر الـ struct عشان حجمه يبقى مضاعف لأكبر alignment فيه.

#include #include #include struct bad { /* careless ordering */ char a; /* 1 */ uint64_t b; /* 8, needs 8-alignment */ char c; /* 1 */ }; struct good { /* largest first */ uint64_t b; char a; char c; }; #pragma pack(push, 1) /* no padding: exactly like on-disk formats */ struct packed { char a; uint64_t b; char c; }; #pragma pack(pop) int main(void) { printf("bad : size %2zu | a@%zu b@%zu c@%zu\n", sizeof(struct bad), offsetof(struct bad, a), offsetof(struct bad, b), offsetof(struct bad, c)); printf("good : size %2zu | b@%zu a@%zu c@%zu\n", sizeof(struct good), offsetof(struct good, b), offsetof(struct good, a), offsetof(struct good, c)); printf("packed : size %2zu | a@%zu b@%zu c@%zu\n", sizeof(struct packed), offsetof(struct packed, a), offsetof(struct packed, b), offsetof(struct packed, c)); return 0; }
struct bad — 24 byte بدل 10
+0x00achar (1 byte)
+0x01 … +0x07pad × 7padding عشان b يبدأ على مضاعف 8
+0x08 … +0x0Fbuint64_t (8 bytes)
+0x10cchar
+0x11 … +0x17pad × 7tail padding: الحجم لازم مضاعف 8 (عشان arrays من الـ struct)

الـ Padding بيكشف الـ layout

لو شفت field عند offset 4 حجمه WORD، وبعده field عند offset 24 حجمه QWORD — يبقى فيه حاجة (أو padding) بين 6 و24. لما تبني struct في Ghidra وتلاقي ثغرات بين الـ offsets المعروفة، املاها undefined لحد ما تلاقي كود بيوصلها. وفي الـ file formats (PE headers، protocols) الـ structs بتكون packed — عشان كده لازم تعرّفها بـ #pragma pack(1) وأنت بتكتب parser.

إعادة بناء struct من الـ Assembly struct recovery

#include struct implant { uint32_t id; uint16_t port; char host[16]; uint64_t sleep_ms; struct implant *next; }; uint64_t total_sleep(const struct implant *p) { uint64_t sum = 0; while (p) { if (p->port == 443) sum += p->sleep_ms; p = p->next; } return sum; } total_sleep: xor eax, eax ; sum = 0 test rdi, rdi ; p == NULL ? je .L5 .L4: cmp WORD PTR [rdi+4], 443 ; p->port (+4, 2 bytes) jne .L3 add rax, QWORD PTR [rdi+24] ; p->sleep_ms (+24, 8 bytes) .L3: mov rdi, QWORD PTR [rdi+32] ; p = p->next (+32, ptr) test rdi, rdi jne .L4 ret .L5: ret

تخيّل إنك شايف الـ Assembly بس. تستنتج: الـ argument pointer لـ struct فيه: WORD عند +4 بيتقارن بـ 443 (port!)، QWORD عند +24 بيتجمع (counter أو مدة)، وpointer عند +32 بيتحط في نفس الـ register وبيتكرر لحد NULL = linked list و+32 هو next. يبقى الحجم على الأقل 40 byte. ده بالظبط اللي بتعمله في Ghidra: Auto Create Structure وبعدين تسمي الـ fields.

Linked Lists — وأسلوب Windows LIST_ENTRY

Windows (والـ kernel بتاع Linux) بيستخدموا أسلوب ذكي: الـ links (Flink/Blink) struct صغير جوه الـ object، ومنه بترجع لأول الـ object بطرح الـ offset — الـ macro اسمه CONTAINING_RECORD. ده بالظبط إزاي الـ malware بيمشي على قايمة الـ DLLs المحملة في الـ PEB (InLoadOrderModuleList).

#include #include typedef struct LIST_ENTRY { struct LIST_ENTRY *Flink, *Blink; } LIST_ENTRY; /* Get the outer struct from a pointer to one of its fields */ #define CONTAINING_RECORD(addr, type, field) \ ((type *)((char *)(addr) - offsetof(type, field))) typedef struct { unsigned base; /* fake "DllBase" */ LIST_ENTRY links; /* NOT at offset 0 on purpose */ const char *name; } MODULE; int main(void) { MODULE mods[3] = { { 0x1000, {0}, "ntdll.dll" }, { 0x2000, {0}, "kernel32.dll" }, { 0x3000, {0}, "user32.dll" } }; LIST_ENTRY head; /* build a circular doubly linked list: head <-> m0 <-> m1 <-> m2 <-> head */ LIST_ENTRY *prev = &head; for (int i = 0; i < 3; i++) { prev->Flink = &mods[i].links; mods[i].links.Blink = prev; prev = &mods[i].links; } prev->Flink = &head; head.Blink = prev; for (LIST_ENTRY *e = head.Flink; e != &head; e = e->Flink) { MODULE *m = CONTAINING_RECORD(e, MODULE, links); printf("0x%04X %s\n", m->base, m->name); } printf("offsetof(MODULE, links) = %zu\n", offsetof(MODULE, links)); return 0; }
🔬
شكلها في الـ disassembly

هتلاقي mov rax, [rax] (اتبع Flink) جوه loop بيتقارن بالـ head، وبعدها lea rcx, [rax-10h] أو قراية بـ offset سالب — ده الـ CONTAINING_RECORD. والـ shellcode بيبدأ غالبًا بـ mov rax, gs:[60h] (عنوان الـ PEB على x64) ← Ldr ← قايمة الـ modules، عشان يلاقي kernel32.dll من غير أي import.

Unions: نفس الذاكرة بأكتر من شكل union

الـ union كل fields بتاعته بتبدأ من نفس العنوان (offset 0)، وحجمه = حجم أكبر field. استخداماته: قراية نفس البيانات بأكتر من نوع، وعمل tagged union (variant).

#include #include typedef union { uint32_t value; uint8_t bytes[4]; struct { uint16_t low, high; } half; } Reg32; typedef enum { VAL_INT, VAL_STR, VAL_IP } ValType; typedef struct { /* tagged union: tag says which member is valid */ ValType type; union { int i; const char *s; uint8_t ip[4]; } as; } ConfigValue; static void show(const ConfigValue *v) { switch (v->type) { case VAL_INT: printf("int %d\n", v->as.i); break; case VAL_STR: printf("str %s\n", v->as.s); break; case VAL_IP: printf("ip %u.%u.%u.%u\n", v->as.ip[0], v->as.ip[1], v->as.ip[2], v->as.ip[3]); break; } } int main(void) { Reg32 r = { .value = 0xCAFEBABE }; printf("value=0x%08X bytes[0]=0x%02X low=0x%04X high=0x%04X size=%zu\n", r.value, r.bytes[0], r.half.low, r.half.high, sizeof r); ConfigValue cfg[] = { { VAL_INT, { .i = 60 } }, { VAL_STR, { .s = "Mozilla/5.0" } }, { VAL_IP, { .ip = { 192, 168, 56, 101 } } }, }; for (int i = 0; i < 3; i++) show(&cfg[i]); return 0; }
💡
Union = بصمة في Windows

LARGE_INTEGER هو union بين QuadPart (64-bit) و{LowPart, HighPart}. ولما Ghidra يشوف نفس الـ offset بيتقرا مرة كـ int ومرة كـ pointer حسب قيمة field تاني، فكّر فورًا في tagged union (زي VARIANT في COM).

Enums و Bit-fields enum · flags

#include enum state { ST_INIT, ST_CONNECT, ST_BEACON = 10, ST_SLEEP, ST_EXIT = 0xFF }; enum perm { /* flag-style enum: combine with | */ PERM_READ = 1 << 0, PERM_WRITE = 1 << 1, PERM_EXEC = 1 << 2, }; struct tcp_flags { /* bit-fields: layout is compiler-specific */ unsigned fin : 1; unsigned syn : 1; unsigned rst : 1; unsigned psh : 1; unsigned ack : 1; }; int main(void) { printf("ST_INIT=%d ST_CONNECT=%d ST_BEACON=%d ST_SLEEP=%d ST_EXIT=%d\n", ST_INIT, ST_CONNECT, ST_BEACON, ST_SLEEP, ST_EXIT); int p = PERM_READ | PERM_EXEC; printf("perm=%d -> %s%s%s\n", p, (p & PERM_READ) ? "r" : "-", (p & PERM_WRITE) ? "w" : "-", (p & PERM_EXEC) ? "x" : "-"); struct tcp_flags f = { .syn = 1, .ack = 1 }; printf("syn=%u ack=%u rst=%u sizeof=%zu\n", f.syn, f.ack, f.rst, sizeof f); return 0; }

الـ enums بتختفي في الـ binary وبتبقى أرقام عادية. لما تلاقي cmp eax, 0Ah وبعدها cmp eax, 0Bh في state machine، اعمل enum في Ghidra/IDA وطبّقه — الكود هيتقرا if (state == ST_BEACON). ونفس الكلام للـ Windows constants (IDA بيعرفها بالـ M key / Ghidra بـ Set Equate).

Flexible Array Member variable-size records

#include #include #include #include typedef struct { uint16_t type; uint16_t length; /* number of bytes in data[] */ uint8_t data[]; /* flexible array member: must be last */ } Record; static Record *make_record(uint16_t type, const void *payload, uint16_t len) { Record *r = malloc(sizeof *r + len); /* header + payload in ONE block */ if (!r) return NULL; r->type = type; r->length = len; memcpy(r->data, payload, len); return r; } int main(void) { Record *r = make_record(0x0003, "whoami", 6); if (!r) return 1; printf("sizeof(Record) = %zu (header only)\n", sizeof(Record)); printf("type=%u length=%u data=%.*s\n", r->type, r->length, r->length, (char *)r->data); free(r); return 0; }

النمط ده (header فيه طول + بيانات بعده مباشرة) هو شكل أغلب الـ network packets والـ resources والـ configs اللي هتقابلها. TLV (Type-Length-Value) بالظبط.

Compound Literals و Designated Initializers و Alignment C99 · C11

3 أدوات حديثة هتشوفها في أي كود C معاصر: compound literal (Type){ … } = object من غير اسم بتعمله في مكانه، وdesignated initializers للـ arrays [index] = value، وalignas/alignof للتحكم في الـ alignment (من <stdalign.h>، وفي C23 بقوا keywords).

#include #include #include typedef struct { int x, y; } Point; static int dist2(Point a, Point b) { int dx = a.x - b.x, dy = a.y - b.y; return dx * dx + dy * dy; } int main(void) { /* compound literals: unnamed objects created in place */ printf("dist2 = %d\n", dist2((Point){ 0, 0 }, (Point){ .x = 3, .y = 4 })); int *p = (int[]){ 10, 20, 30 }; /* a temporary array */ printf("p[2] = %d\n", p[2]); /* designated initializers for arrays: index -> value, the rest is 0/NULL */ const char *status[8] = { [0] = "OK", [2] = "NOT_FOUND", [5] = "IO_ERROR" }; for (int i = 0; i < 8; i++) if (status[i] != NULL) printf("status[%d] = %s\n", i, status[i]); /* alignment */ alignas(16) unsigned char key[16] = { 0 }; /* e.g. for SSE / AES-NI loads */ printf("alignof(double) = %zu, alignof(Point) = %zu\n", alignof(double), alignof(Point)); printf("key is 16-byte aligned: %s\n", ((uintptr_t)key % 16 == 0) ? "yes" : "no"); return 0; }

في الـ binary

Array بـ designated initializers زي status[] بيبقى جدول في .rdata فيه فجوات أصفار بين الـ pointers، وده شكل جداول الأخطاء والأوامر اللي الـ index فيها هو الـ command ID. والـ alignas(16) على متغير محلي بيخلي الـ prologue فيه and rsp, -16 (أو 0FFFFFFFFFFFFFFF0h)، وبيظهر كتير مع كود crypto بيستخدم SSE/AES-NI (movdqa، aesenc).
