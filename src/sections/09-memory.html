خريطة الذاكرة والـ Dynamic Memory

كل process ليه مساحة عناوين متقسمة لمناطق: الكود، البيانات الثابتة، الـ globals، الـ heap، الـ stack، والـ DLLs. لما تفتح x64dbg وتشوف عنوان، أول سؤال: العنوان ده في أنهي منطقة؟ الإجابة بتقولك تقريبًا كل حاجة عن البيانات اللي فيه.

شغّله كذا مرة ولاحظ: عناوين الـ heap والـ stack بتتغير كل مرة (ASLR)، لكن الكود والـ globals ثابتين عند 0x40xxxx — لأن Compiler Explorer بيبني executable non-PIE (مش Position Independent)، فالـ image بيتحمل دايمًا على نفس العنوان. مع PIE (الافتراضي في أغلب توزيعات Linux، وزي /DYNAMICBASE على Windows) الـ image نفسه بيتنقل كمان. ولاحظ إن الكود والـ globals قريبين من بعض (نفس الـ image)، والـ stack بعيد جدًا فوق (0x7ff…). من شكل العنوان بس تقدر تخمّن منطقته.

Hash Table: بحث سريع بالمفتاح hashing · chaining

الـ dynamic array كويس لما تمشي على العناصر بالترتيب. لكن لو عايز تسأل "الـ domain ده شفته قبل كده؟" بسرعة وسط آلاف، محتاج hash table: دالة hash بتحوّل المفتاح لرقم، والرقم بيحدد bucket، وكل bucket فيه linked list صغيرة للمفاتيح اللي وقعت فيه (chaining).

#include #include #include #include #define BUCKETS 16 /* power of 2 -> index = hash & 15 */ typedef struct Node { char *key; int count; struct Node *next; /* collisions form a list */ } Node; static Node *table[BUCKETS]; static uint32_t fnv1a(const char *s) { /* FNV-1a, 32-bit */ uint32_t h = 2166136261u; /* offset basis 0x811C9DC5 */ while (*s) { h ^= (uint8_t)*s++; h *= 16777619u; /* FNV prime 0x01000193 */ } return h; } static void add(const char *key) { uint32_t b = fnv1a(key) & (BUCKETS - 1); for (Node *n = table[b]; n != NULL; n = n->next) if (strcmp(n->key, key) == 0) { n->count++; return; } Node *n = malloc(sizeof *n); if (n == NULL) return; n->key = malloc(strlen(key) + 1); if (n->key == NULL) { free(n); return; } strcpy(n->key, key); n->count = 1; n->next = table[b]; /* push at the head of the bucket */ table[b] = n; } static int count_of(const char *key) { for (Node *n = table[fnv1a(key) & (BUCKETS - 1)]; n != NULL; n = n->next) if (strcmp(n->key, key) == 0) return n->count; return 0; } int main(void) { const char *iocs[] = { "evil.example", "10.0.0.5", "evil.example", "update.bin", "10.0.0.5", "evil.example" }; for (size_t i = 0; i < sizeof iocs / sizeof iocs[0]; i++) add(iocs[i]); uint32_t h = fnv1a("evil.example"); printf("fnv1a(\"evil.example\") = 0x%08X -> bucket %u\n", h, h & (BUCKETS - 1)); printf("evil.example seen %d times\n", count_of("evil.example")); printf("10.0.0.5 seen %d times\n", count_of("10.0.0.5")); printf("clean.org seen %d times\n", count_of("clean.org")); for (int b = 0; b < BUCKETS; b++) { /* free everything */ Node *n = table[b]; while (n != NULL) { Node *next = n->next; free(n->key); free(n); n = next; } } return 0; }

Binary Search Tree trees

شجرة كل node فيها مفتاح، وكل اللي على الشمال أصغر وكل اللي على اليمين أكبر. البحث والإضافة متوسطهم O(log n)، والمشي in-order (شمال ← النود ← يمين) بيطلع المفاتيح مترتبة. ده أساس std::map في C++ (المرحلة 22) والـ RTL_AVL_TABLE في Windows.

"yes" : "no", contains(root, 5) ? "yes" : "no"); destroy(root); return 0; }
