الـ Pointers: متغيرات بتشاور على متغيرات

الـ pointer مجرد رقم — عنوان في الذاكرة (8 bytes على x64). مفيش سحر. لكن الفهم العميق ليه هو اللي بيفرق بين اللي بيقرا الـ disassembly واللي بيتوه فيه، لأن الـ Assembly كله عناوين: كل [rax] و[rbp-8] هي pointer dereference.

lea vs mov: أهم فرق في الـ Assembly address vs value

void set_to_42(int *p) { *p = 42; } int demo(void) { int x = 7; int *ptr = &x; set_to_42(ptr); return *ptr; } set_to_42: push rbp mov rbp, rsp mov QWORD PTR [rbp-8], rdi ; save p (8 bytes) mov rax, QWORD PTR [rbp-8] ; rax = p mov DWORD PTR [rax], 42 ; *p = 42 nop pop rbp ret demo: push rbp mov rbp, rsp sub rsp, 16 mov DWORD PTR [rbp-12], 7 ; x = 7 lea rax, [rbp-12] ; rax = &x (ADDRESS) mov QWORD PTR [rbp-8], rax ; ptr = &x mov rax, QWORD PTR [rbp-8] mov rdi, rax ; 1st arg = ptr call set_to_42 mov rax, QWORD PTR [rbp-8] ; rax = ptr mov eax, DWORD PTR [rax] ; eax = *ptr (VALUE) leave ret

القاعدة الذهبية

lea reg, [X] = العنوان X نفسه (&) — مفيش قراية من الذاكرة. mov reg, [X] = المحتوى اللي في X (*). لما تشوف lea rcx, [rbp-40h] قبل call، يبقى الدالة واخدة pointer لمتغير محلي (غالبًا buffer أو struct هتملاه — output parameter). والقوسين [ ] في Intel syntax = dereference دايمًا (ماعدا مع lea).

نوع الـ Pointer بيحدد طريقة القراية types

العنوان مجرد رقم، لكن نوع الـ pointer بيقول للـ compiler: تقرا كام byte، وتفسرهم إزاي، وتزوّد كام لما تعمل p + 1. ونفس الذاكرة ممكن تتقرا بأكتر من نوع:

#include #include #include int main(void) { uint32_t value = 0x41424344; /* "DCBA" in memory (little-endian) */ unsigned char *bytes = (unsigned char *)&value; /* always allowed */ printf("as bytes : %02X %02X %02X %02X\n", bytes[0], bytes[1], bytes[2], bytes[3]); printf("as chars : %c%c%c%c\n", bytes[0], bytes[1], bytes[2], bytes[3]); /* Reading as another type: use memcpy (avoids strict-aliasing UB) */ uint16_t low_half; memcpy(&low_half, &value, sizeof low_half); printf("low u16 : 0x%04X\n", low_half); /* Pointer arithmetic is scaled by the pointed-to type */ uint32_t arr[3] = { 0x11111111, 0x22222222, 0x33333333 }; uint32_t *pi = arr; unsigned char *pc = (unsigned char *)arr; printf("pi + 1 moves %td bytes, pc + 1 moves %td byte\n", (char *)(pi + 1) - (char *)pi, (pc + 1) - pc); printf("*(pi + 2) = 0x%08X\n", *(pi + 2)); return 0; }
⚠️
Strict Aliasing

قراية ذاكرة بنوع غير نوعها الأصلي عن طريق cast (مثلًا *(float *)&my_int) = UB في C، والـ optimizer ممكن يكسر كودك. الاستثناء المضمون: char/unsigned char يقدروا يقروا أي حاجة. للتحويل بين الأنواع استخدم memcpy — والـ compiler بيشيلها ويعملها instruction واحدة.

const مع الـ Pointers const correctness

التعريفتقدر تغير القيمة *p؟تقدر تغير الـ pointer p؟
int *p✅✅
const int *p (= int const *p)❌✅
int *const p✅❌
const int *const p❌❌

اقرأها من اليمين للشمال: int *const p = "p is a const pointer to int". وفي التوقيعات: const char *src معناها الدالة مش هتعدّل البيانات — معلومة مفيدة جدًا وأنت بتحلل (input vs output buffers).
