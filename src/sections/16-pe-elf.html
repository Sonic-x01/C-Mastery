من Object File لـ Executable: الـ Linking وصيغة PE (وELF)

الـ executable مش مجرد كود — هو ملف منظم فيه headers بتقول للـ loader: حمّلني فين، ابدأ منين، أنهي DLLs أحتاجها، وأنهي أجزاء قابلة للتنفيذ. أول 5 دقايق في تحليل أي sample بتبقى على الـ headers دي (في PE-bear أو pestudio أو DIE) — ومنها بتعرف هل هو packed، بيستورد إيه، اتبنى إمتى وبإيه.

Object Files والـ Symbols والـ Relocations linking

كل .c بيتحول لـ object file فيه: الكود، البيانات، جدول symbols (الدوال والمتغيرات اللي معرّفها أو محتاجها)، وrelocations (أماكن في الكود لسه عنوانها مش معروف). الـ linker بيجمع الـ objects، بيحل كل symbol، وبيكتب العناوين النهائية.

$ gcc -c crypto.c && nm crypto.o 0000000000000000 B crypto_calls # B = .bss global 0000000000000040 T crc32 # T = exported function in .text 0000000000000000 T xor_crypt 0000000000000000 t build_table # t = static (local) function 0000000000000000 b table # b = static data in .bss U memset # U = undefined: must come from elsewhere $ objdump -dr -M intel crypto.o | head # -r shows the relocations

RVA vs File Offset address translation

RVA (Relative Virtual Address) = offset من الـ ImageBase في الذاكرة بعد التحميل. لكن في الملف على الـ disk، الـ sections مترتبة بـ alignment مختلف (FileAlignment عادة 0x200 بدل SectionAlignment 0x1000). عشان تقرا حاجة من الملف بالـ RVA بتاعها، لازم تحولها:

file_offset = RVA − section.VirtualAddress + section.PointerToRawData

#include #include typedef struct { char name[9]; uint32_t va, vsize, raw_ptr, raw_size; } Section; /* Typical layout of a small 64-bit PE */ static const Section sections[] = { { ".text", 0x1000, 0x1A2C, 0x0400, 0x1C00 }, { ".rdata", 0x3000, 0x0B14, 0x2000, 0x0C00 }, { ".data", 0x4000, 0x0640, 0x2C00, 0x0200 }, { ".bss", 0x5000, 0x0180, 0x0000, 0x0000 }, /* exists only in memory */ }; static long rva_to_offset(uint32_t rva) { for (size_t i = 0; i < sizeof sections / sizeof sections[0]; i++) { const Section *s = §ions[i]; if (rva >= s->va && rva < s->va + s->vsize) { if (rva - s->va >= s->raw_size) { printf(" (RVA 0x%X is in %s but has no file data)\n", rva, s->name); return -1; } return (long)(rva - s->va + s->raw_ptr); } } return -1; } int main(void) { uint32_t rvas[] = { 0x1000, 0x14F0, 0x3020, 0x4010, 0x5004, 0x9000 }; for (int i = 0; i < 6; i++) { long off = rva_to_offset(rvas[i]); if (off >= 0) printf("RVA 0x%04X -> file offset 0x%04lX\n", rvas[i], (unsigned long)off); else printf("RVA 0x%04X -> not in file\n", rvas[i]); } return 0; }

الـ Imports: IAT و INT import table

DataDirectory[1] بيشاور على array من IMAGE_IMPORT_DESCRIPTOR (20 byte لكل DLL) بينتهي بـ entry كلها أصفار. كل descriptor فيه:

FieldOffsetالمعنى
OriginalFirstThunk0RVA للـ INT (Import Name Table): أسماء الدوال (أو ordinals) — مبتتغيرش
TimeDateStamp / ForwarderChain4 / 8للـ binding (غالبًا 0)
Name12RVA لاسم الـ DLL ("KERNEL32.dll")
FirstThunk16RVA للـ IAT: في الملف نسخة من الـ INT، والـ loader بيكتب فوقها العناوين الحقيقية
🔬
كل call لـ API = قراية من الـ IAT

في الكود هتشوف call qword ptr [rip+2F8Ah] — ده indirect call على خانة في الـ IAT. IDA بيكتبها call cs:__imp_CreateFileW. والـ Import Table نفسها أقوى "ملخص" للـ binary: CryptEncrypt + FindFirstFileW = ransomware؟ WriteProcessMemory + CreateRemoteThread = injection؟ InternetOpenUrlA = تواصل شبكة. أداة capa بتعمل التحليل ده أوتوماتيك. وتقنية imphash (hash للـ imports) بتربط samples من نفس العيلة.

TLS Callbacks: كود قبل الـ Entry Point anti-analysis

DataDirectory[9] (TLS) ممكن يحتوي على callbacks بيتنفذوا قبل الـ entry point. الـ malware بيستخدمها عشان يعمل anti-debugging أو يفك نفسه قبل ما الـ debugger يوقف عند الـ entry. في x64dbg فعّل "Break on TLS Callbacks" من الـ settings، وفي IDA هتلاقيها في قايمة الـ entry points.

#include #include #include #include /* Shannon entropy in bits per byte: 0.0 (all same) ... 8.0 (perfectly random) */ static double entropy(const uint8_t *data, size_t len) { if (len == 0) return 0.0; size_t counts[256] = { 0 }; for (size_t i = 0; i < len; i++) counts[data[i]]++; double h = 0.0; for (int b = 0; b < 256; b++) { if (!counts[b]) continue; double p = (double)counts[b] / (double)len; h -= p * log2(p); } return h; } int main(void) { static uint8_t zeros[4096], text[4096], prng[4096]; const char *s = "This program cannot be run in DOS mode. kernel32.dll GetProcAddress "; for (size_t i = 0; i < sizeof text; i++) text[i] = (uint8_t)s[i % strlen(s)]; uint32_t x = 0x12345678; /* xorshift32: "encrypted-looking" bytes */ for (size_t i = 0; i < sizeof prng; i++) { x ^= x << 13; x ^= x >> 17; x ^= x << 5; prng[i] = (uint8_t)x; } printf("zeros : %.3f bits/byte\n", entropy(zeros, sizeof zeros)); printf("text : %.3f bits/byte\n", entropy(text, sizeof text)); printf("random : %.3f bits/byte <- looks packed / encrypted\n", entropy(prng, sizeof prng)); return 0; }

✏️ تمارين المرحلة 16

عشان توصل للـ NT headers من أول ملف PE، بتقرا... أول 4 bytes الـ offset الثابت 0x80 دايمًا قيمة e_lfanew (DWORD عند offset 0x3C) — دي الـ offset بتاع "PE\0\0" آخر الملف الـ e_lfanew بيتغير من ملف للتاني (حسب حجم الـ DOS stub والـ Rich header)، فلازم تقراه. فيه field في الـ DOS header بيشاور على الـ PE header. Section: VirtualAddress=0x3000، PointerToRawData=0x2400. الـ RVA 0x3150 = أنهي file offset؟ 0x2550 0x3150 0x5550 0x0150 0x3150 − 0x3000 + 0x2400 = 0x2550. طبّق: RVA − VirtualAddress + PointerToRawData. Section اسمها UPX0 بـ SizeOfRawData = 0 وVirtualSize = 0x28000 وصلاحيات RWX. ده بيقول إيه؟ Section بايظة Section للـ resources .bss عادية الـ binary معمول له pack بـ UPX: المساحة دي فاضية في الملف وهتتملا بالكود الأصلي بعد ما الـ stub يفكه UPX بيعمل section فاضية (UPX0) للكود المفكوك، وUPX1 فيها البيانات المضغوطة + الـ stub. upx -d بيفكه — إلا لو اتعدّل عشان يكسر الأداة. اسم الـ section نفسه دليل، ومع الأحجام والصلاحيات… إيه الفرق بين الـ INT والـ IAT في الملف قبل التحميل وبعده؟ مفيش فرق أبدًا قبل التحميل الاتنين فيهم أسماء/ordinals. بعد التحميل الـ loader بيكتب العناوين الحقيقية للدوال في الـ IAT، والـ INT بيفضل زي ما هو الـ INT فيه عناوين والـ IAT فيه أسماء الـ IAT مش موجود في الملف عشان كده لما بتعمل dump لـ process من الذاكرة (بعد unpacking)، لازم "تصلح" الـ IAT (بأدوات زي Scylla) عشان الملف يتفتح صح في الأدوات. مين اللي بيملا العناوين الحقيقية، وفين؟ لقيت sample الـ TimeDateStamp بتاعه بيقول 2039. الاستنتاج؟ الـ sample من المستقبل الـ compiler بايظ الـ timestamp متزور (عمدًا أو بأداة)، فمتعتمدش عليه في الـ timeline الملف مش PE الـ timestamp سهل التعديل. (وكمان الـ builds الـ reproducible في MSVC بتحط hash مكانه.) قارنه بمصادر تانية: debug directory، الـ Rich header، وتاريخ أول ظهور في VirusTotal. هل الـ headers ممكن تتعدل بعد الـ build؟ اكتب دالة بتحول Characteristics لـ string صلاحيات RWX (0x40000000=R، 0x80000000=W، 0x20000000=X، وحط - لو مش موجودة)، واطبع لكل section اسمها وصلاحياتها، وكلمة SUSPICIOUS لو فيها W وX مع بعض:
