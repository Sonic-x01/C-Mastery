المرحلة الحادية عشر

الـ Preprocessor، المشاريع متعددة الملفات، والـ Build

قبل الـ compile، الـ preprocessor بيعمل نسخ ولصق نصي: بيدخّل الـ headers، بيستبدل الـ macros، وبيشيل أجزاء من الكود حسب الشروط. كل ده بيختفي من الـ binary — لكن بيسيب آثار مهمة جدًا للمحلل: مسارات ملفات، أسماء دوال، رسايل debug، وأحيانًا مسار الـ PDB بتاع جهاز الكاتب نفسه.

#define: ثوابت و Macros macros

#include #define BUF_SIZE 256 #define C2_PORT 443 #define SQUARE_BAD(x) x * x /* BUG: no parentheses */ #define SQUARE(x) ((x) * (x)) /* correct */ #define MAX(a, b) ((a) > (b) ? (a) : (b)) #define ARRAY_LEN(a) (sizeof(a) / sizeof((a)[0])) int main(void) { printf("BUF_SIZE=%d C2_PORT=%d\n", BUF_SIZE, C2_PORT); printf("SQUARE_BAD(2+3) = %d (expands to 2+3*2+3)\n", SQUARE_BAD(2 + 3)); printf("SQUARE(2+3) = %d\n", SQUARE(2 + 3)); int i = 5; int m = MAX(i++, 3); /* DANGER: i++ is evaluated twice */ printf("MAX(i++,3) = %d, i is now %d (not 6!)\n", m, i); int ports[] = { 80, 443, 8080, 8443 }; printf("ARRAY_LEN(ports) = %zu\n", ARRAY_LEN(ports)); return 0; }
⚠️
قواعد الـ macros الآمنة

(1) حط أقواس حوالين كل parameter وحوالين الـ macro كله. (2) متبعتش arguments ليها side effects (i++، استدعاء دالة). (3) لو ينفع، استخدم static inline function أو enum/const بدل الـ macro — ليها type checking وبتظهر في الـ debugger.

Conditional Compilation #if · #ifdef

#include /* #define DEBUG_BUILD */ /* uncomment to enable logging */ #ifdef DEBUG_BUILD #define LOG(...) printf("[dbg] " __VA_ARGS__) #else #define LOG(...) ((void)0) /* vanishes completely in release */ #endif int main(void) { #if defined(_WIN32) puts("platform: Windows"); #elif defined(__linux__) puts("platform: Linux"); #elif defined(__APPLE__) puts("platform: macOS"); #else puts("platform: unknown"); #endif #if defined(__x86_64__) || defined(_M_X64) puts("arch: x86-64"); #endif LOG("this only exists in debug builds\n"); printf("compiled by GCC %d.%d, C standard %ld\n", __GNUC__, __GNUC_MINOR__, __STDC_VERSION__); return 0; }
🔬
نسخة الـ debug من الـ malware = هدية

الـ malware authors أحيانًا بيسيبوا LOG(...) شغالة بالغلط، أو بيرفعوا نسخة debug. ساعتها الـ binary بيبقى مليان رسايل زي "[+] injecting into %d" بتشرحلك كل خطوة. ولما الـ platform checks بتختلف، هتلاقي samples لنفس العيلة على Windows وLinux بكود مختلف شوية.

مشروع متعدد الملفات .h + .c

الـ header (.h) فيه declarations بس (الواجهة)، والـ source (.c) فيه الـ definitions. كل .c بيتعمله compile لوحده لـ object file، وبعدين الـ linker بيربطهم.

#ifndef CRYPTO_H /* include guard: prevents double inclusion */ #define CRYPTO_H /* (or simply: #pragma once) */ #include #include void xor_crypt(uint8_t *buf, size_t len, const uint8_t *key, size_t key_len); uint32_t crc32(const uint8_t *data, size_t len); extern int crypto_calls; /* declared here, DEFINED in crypto.c */ #endif #include "crypto.h" /* quotes: search the project folder first */ int crypto_calls = 0; /* the one definition */ static uint32_t table[256]; /* static: private to this file */ static int table_ready = 0; static void build_table(void) { /* ... */ table_ready = 1; } void xor_crypt(uint8_t *buf, size_t len, const uint8_t *key, size_t key_len) { crypto_calls++; for (size_t i = 0; i < len; i++) buf[i] ^= key[i % key_len]; } uint32_t crc32(const uint8_t *data, size_t len) { if (!table_ready) build_table(); /* ... */ (void)data; (void)len; return 0; } # Compile each .c into an object file, then link gcc -c crypto.c -o crypto.o gcc -c main.c -o main.o gcc main.o crypto.o -o tool # Static library (.a / .lib): code is COPIED into the executable ar rcs libcrypto_mini.a crypto.o gcc main.c -L. -lcrypto_mini -o tool # Shared library (.so / .dll): code is loaded at run time gcc -shared -fPIC crypto.c -o libcrypto_mini.so # See the symbols: T = defined function, U = undefined (imported), t = static nm crypto.o

مع #define DOUBLE(x) x + x، قيمة 3 * DOUBLE(2)؟ 12 8 10 6 بتتحول لـ 3 * 2 + 2 = 8. الحل: #define DOUBLE(x) ((x) + (x)). الـ macro استبدال نصي حرفي. اكتب التعبير بعد الاستبدال واحسبه بالـ precedence. وظيفة #ifndef X_H / #define X_H / ... / #endif في أول الـ header؟ بتسرّع التنفيذ بتمنع الـ linker من التكرار بتعرّف متغيرات global Include guard: بتمنع محتوى الـ header يتضاف مرتين في نفس الـ translation unit أول مرة X_H مش متعرّف فالمحتوى بيتضاف ويتعرّف. أي #include تاني بيلاقيه متعرّف فيتخطاه. فكّر إيه اللي بيحصل لو header اتعمل له include من ملفين بيتعمل لهم include في نفس الـ .c. لقيت في sample string زي C:\Users\kaz\source\repos\Stealer\x64\Release\Stealer.pdb. ده جه منين؟ الـ debug directory في الـ PE اللي MSVC بيكتب فيها مسار ملف الـ PDB وقت الـ build الـ malware بيطبعه للمستخدم من Windows نفسه من الـ antivirus الـ PDB path معلومة attribution قوية: اسم المستخدم، اسم المشروع، وهيكل الفولدرات. كتير من الـ reports بتربط samples ببعض عن طريقه. MSVC بيكتب حاجة عن ملف الـ debug symbols جوه الـ EXE. دالة متعرّفة static في crypto.c اسمها build_table. ملف main.c حاول يناديها. النتيجة؟ بتشتغل عادي Runtime crash Linker error: undefined reference — لأن static بيخليها internal للملف بتاعها الـ preprocessor بيرفضها static على مستوى الملف = internal linkage. الدالة مش ظاهرة لأي object file تاني. إيه معنى static لدالة برا أي block؟ binary حجمه 3 MB وفيه آلاف الدوال، والـ imports قليلة جدًا. أول خطوة منطقية في IDA/Ghidra؟ تحلل كل دالة بالترتيب تطبّق library signatures (FLIRT / Function ID) عشان تتعرف على كود المكتبات الـ statically linked وتركّز على كود الكاتب تمسح الـ binary تدوّر على main بالـ strings أغلب الكود في binary زي ده غالبًا مكتبات (CRT، OpenSSL، curl…). الـ signatures بتسمّيها أوتوماتيك وبتوفر عليك ساعات.
