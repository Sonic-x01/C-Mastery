المرحلة الثانية

المتغيرات، الأنواع، والبيانات زي ما هي في الذاكرة

في C، المتغير = مكان في الذاكرة له حجم معيّن، والنوع بيقول للـ compiler: الـ bytes دي أقراها إزاي. نفس الـ 4 bytes ممكن تبقى int أو float أو 4 حروف — الذاكرة مش عارفة الفرق. ده أهم درس في الـ RE كله.

Bits وBytes وHex الأساس

الـ bit = 0 أو 1. الـ byte = 8 bits = قيمة من 0 لـ 255. الـ Hex (أساس 16) هو اللغة الرسمية للـ Reverser لأن كل خانة hex = 4 bits بالظبط، فكل byte = خانتين hex: 0xFF = 1111 1111 = 255.

LP64 vs LLP64 — فخ حقيقي

على Windows 64-bit، long لسه 4 bytes. عشان كده الـ Windows API بتستخدم DWORD (= unsigned long = 32-bit) وDWORD_PTR/ULONG_PTR للقيم اللي بحجم pointer. لما تحلل sample، الحجم في الـ Assembly (DWORD PTR vs QWORD PTR) هو اللي بيقولك النوع الحقيقي — مش اسم النوع.

Signed vs Unsigned و Two's Complement negative numbers

الـ CPU مش بيعرف "سالب". الأرقام السالبة بتتخزن بنظام Two's Complement: عشان تجيب -x: اقلب كل الـ bits (~x) وزوّد 1. النتيجة: نفس الـ bits ليها معنيين حسب ما تقراها signed ولا unsigned.

8-bit two's complement — نفس الـ bits، قراءتين
0000 00010x01unsigned = 1 · signed = 1
0111 11110x7Funsigned = 127 · signed = 127 (أكبر موجب)
1000 00000x80unsigned = 128 · signed = -128 (أصغر سالب)
1111 11100xFEunsigned = 254 · signed = -2
1111 11110xFFunsigned = 255 · signed = -1
قاعدة سريعة: لو أعلى bit (الـ sign bit) = 1، يبقى الرقم سالب في القراءة الـ signed.
#include #include int main(void) { int8_t s = -1; uint8_t u = (uint8_t)s; /* same bits, unsigned view */ printf("int8 -1 as uint8 = %u (0x%02X)\n", u, u); int32_t neg = -2; printf("int32 -2 in hex = 0x%08X\n", (uint32_t)neg); printf("~5 + 1 = %d\n", ~5 + 1); uint32_t big = 0xFFFFFFF6u; printf("0xFFFFFFF6 signed = %d\n", (int32_t)big); return 0; }

الـ Overflow: الـ unsigned بيلف، والـ signed خطر wraparound

  • Unsigned overflow = سلوك محدد: الحساب بيتم modulo 2ⁿ. 255 + 1 في uint8_t = 0، و0 - 1 = 255.
  • Signed overflow = Undefined Behavior! المعيار مبيضمنش أي نتيجة، والـ compiler بيفترض إنه مبيحصلش (وبيعمل optimization على الأساس ده). على الـ hardware غالبًا بيلف، بس متعتمدش على ده أبدًا.
#include #include #include int main(void) { uint8_t counter = 250; for (int i = 0; i < 10; i++) counter++; /* wraps at 256 */ printf("uint8 250 + 10 = %u\n", counter); unsigned int zero = 0; printf("unsigned 0 - 1 = %u\n", zero - 1); /* UINT_MAX */ printf("INT_MAX = %d\n", INT_MAX); printf("INT_MIN = %d\n", INT_MIN); /* Safe check BEFORE adding, never after */ int a = INT_MAX, b = 1; if (b > 0 && a > INT_MAX - b) printf("a + b would overflow -> refused\n"); return 0; }
💥
Integer overflow = ثغرة

لما حساب حجم (مثلًا count * sizeof(item)) يعمل overflow، الناتج بيبقى رقم صغير، فبيتحجز buffer صغير، وبعدين بيتكتب فيه بيانات كتير ← heap overflow. ثغرات كتير حقيقية بدأت كده. اتعلم تشوف الحسابات دي في الـ Assembly قبل malloc.

Endianness: ترتيب الـ bytes في الذاكرة little-endian

x86 و x64 و (افتراضيًا) ARM كلهم Little-Endian: الـ byte الأصغر قيمةً بيتخزن الأول (في العنوان الأصغر). فالرقم 0x11223344 بيتخزن في الذاكرة كده: 44 33 22 11. الـ network protocols بتستخدم Big-Endian (اسمه "network byte order").

uint32_t x = 0x11223344; — في الذاكرة (x86)
&x + 00x44الـ byte الأقل أهمية (LSB) الأول
&x + 10x33
&x + 20x22
&x + 30x11الـ byte الأكثر أهمية (MSB) الآخر
#include #include int main(void) { uint32_t x = 0x11223344; unsigned char *p = (unsigned char *)&x; /* look at the raw bytes */ printf("value : 0x%08X\n", x); printf("bytes :"); for (size_t i = 0; i < sizeof x; i++) printf(" %02X", p[i]); printf("\n"); /* 'MZ' magic of a PE file read as a little-endian WORD */ unsigned char header[2] = { 0x4D, 0x5A }; /* 'M' 'Z' */ uint16_t magic = (uint16_t)(header[0] | (header[1] << 8)); printf("MZ as uint16 = 0x%04X\n", magic); /* 0x5A4D */ return 0; }
🔬
هتشوف ده في كل hex editor

في ملف PE، أول 2 bytes هما 4D 5A ("MZ")، لكن لو قريتهم كـ WORD هيبقوا 0x5A4D — وده بالظبط الثابت IMAGE_DOS_SIGNATURE اللي بتلاقيه في كود الـ malware اللي بيدور على PE headers في الذاكرة: cmp word ptr [rax], 5A4Dh. نفس الكلام للـ strings الصغيرة المخزنة كأرقام: mov eax, 6C6C6568h = "hell" مقلوبة!

char = رقم صغير ASCII

char في C مجرد integer بحجم byte. 'A' هي بالظبط الرقم 65. ده بيخلي العمليات على الحروف حسابات عادية — وده اللي بيعمله أي string decoder في الـ malware.

#include int main(void) { char c = 'A'; printf("'%c' = %d = 0x%X\n", c, c, c); printf("'A' + 2 = '%c'\n", c + 2); printf("'7' - '0' = %d (digit char -> number)\n", '7' - '0'); printf("lower: '%c' ('A' | 0x20)\n", 'A' | 0x20); printf("upper: '%c' ('a' & ~0x20)\n", 'a' & ~0x20); /* Escape sequences are single bytes too */ printf("\\n=%d \\t=%d \\0=%d \\x41='%c'\n", '\n', '\t', '\0', '\x41'); return 0; }
⚠️
char signed ولا unsigned؟

المعيار سايبها للمنصة! على x86 (gcc/MSVC) char = signed، وعلى ARM Linux غالبًا unsigned. لو بتتعامل مع bytes خام (ملفات، shellcode، buffers) استخدم دايمًا unsigned char أو uint8_t — وإلا 0xFF هيبقى -1 ويبوظ مقارناتك.

التحويلات: Truncation و Sign Extension casts

لما تحوّل من نوع كبير لصغير، الـ bits العليا بتتقص (truncation). ولما تحوّل من صغير لكبير: الـ unsigned بيتملا أصفار (zero extension)، والـ signed بيتملا بنسخة من الـ sign bit (sign extension).

#include #include int main(void) { uint32_t big = 0x12345678; uint8_t low = (uint8_t)big; /* truncation: keeps 0x78 */ printf("truncate 0x%08X -> 0x%02X\n", big, low); int8_t sb = (int8_t)0xF0; /* -16 */ uint8_t ub = 0xF0; /* 240 */ int32_t s32 = sb; /* sign-extended -> movsx */ uint32_t u32 = ub; /* zero-extended -> movzx */ printf("sign-extend 0xF0 -> 0x%08X (%d)\n", (uint32_t)s32, s32); printf("zero-extend 0xF0 -> 0x%08X (%u)\n", u32, u32); double d = 3.99; printf("(int)3.99 = %d (truncates toward zero)\n", (int)d); printf("(int)-3.99 = %d\n", (int)-3.99); printf("7 / 2 = %d, 7 / 2.0 = %.1f, 7 %% 2 = %d, -7 / 2 = %d, -7 %% 2 = %d\n", 7 / 2, 7 / 2.0, 7 % 2, -7 / 2, -7 % 2); return 0; }
🔬
movsx vs movzx بيكشفلك النوع

اضغط ⚙ ASM على المثال اللي فوق. لما تشوف movsx (sign-extend) يبقى المتغير الأصلي signed، ولما تشوف movzx يبقى unsigned. ده من أهم الأدلة اللي بتستخدمها عشان تحدد أنواع المتغيرات في Ghidra وتصححها.

فخ المقارنة بين signed و unsigned usual arithmetic conversions

لما تعمل عملية بين int وunsigned int، الـ int بيتحوّل لـ unsigned الأول! فـ -1 بيبقى 4294967295. والأنواع الأصغر من int (char وshort) بتتحول لـ int قبل أي عملية (integer promotion).

#include int main(void) { int a = -1; unsigned int b = 1; if (a < b) printf("-1 < 1u : true\n"); else printf("-1 < 1u : FALSE! (-1 became %u)\n", (unsigned int)a); /* Promotion: chars become int before arithmetic */ unsigned char x = 200, y = 100; int sum = x + y; /* 300, no wrap: computed as int */ unsigned char wrapped = x + y; /* stored back in 8 bits: 44 */ printf("x + y as int = %d, stored in uchar = %d\n", sum, wrapped); return 0; }
🔬
jl ولا jb؟

الـ CPU عنده instructions مقارنة مختلفة: jl/jg/jle/jge للـ signed، وjb/ja/jbe/jae للـ unsigned. لما تشوف jb بعد cmp في check على طول buffer، يبقى الطول unsigned. ولو شفت jl على طول جاي من المستخدم… ممكن تكون لقيت ثغرة: طول سالب هيعدّي الـ check.

Floating Point من جوه IEEE-754

float = 32 bit: 1 sign + 8 exponent + 23 mantissa. double = 64 bit (1 + 11 + 52). الأعداد العشرية زي 0.1 مش بتتمثل بدقة! ومتقارنش floats بـ == أبدًا.

#include #include #include int main(void) { float f = 1.0f; uint32_t bits; memcpy(&bits, &f, sizeof bits); /* safe way to view raw bits */ printf("1.0f -> 0x%08X\n", bits); f = -2.5f; memcpy(&bits, &f, sizeof bits); printf("-2.5f -> 0x%08X\n", bits); double sum = 0.1 + 0.2; printf("0.1 + 0.2 = %.17f\n", sum); printf("0.1 + 0.2 == 0.3 ? %s\n", sum == 0.3 ? "yes" : "no"); return 0; }
🔬
علّم نفسك تعرف الـ floats في الـ hex

0x3F800000 = 1.0f، و0x40000000 = 2.0f، و0x3FF0000000000000 = 1.0 (double). لو شفت ثوابت زي دي في الـ .rdata أو instructions زي movss/movsd/cvtsi2sd، يبقى فيه حسابات float.

التعريف، القيم الابتدائية، و const declarations

#include int global_counter; /* globals are zero-initialised (.bss) */ const int MAX_RETRIES = 3; /* read-only (.rodata) */ int main(void) { int a = 10, b = 0x1F, c = 017; /* decimal, hex, octal (leading 0!) */ unsigned long long big = 0xDEADBEEFCAFEBABEULL; char letter = 'Z'; int local; /* local: NOT initialised -> garbage */ local = 5; /* always assign before use */ printf("%d %d %d\n", a, b, c); printf("0x%llX\n", big); printf("%c %d %d %d\n", letter, global_counter, MAX_RETRIES, local); return 0; }
⚠️
الصفر في الأول = octal!

017 مش 17، ده 15 (octal). غلطة كلاسيكية في كود الـ permissions والـ masks. والمتغيرات المحلية مش متصفّرة — قيمتها أي حاجة كانت في الـ stack (وده ممكن يسرّب بيانات: uninitialized memory disclosure).
