8 مشاريع: أدوات حقيقية بيستخدمها المحللين

المحلل المحترف بيكتب أدواته الصغيرة بنفسه: decoder لـ config، parser لـ format غريب، brute-forcer لـ key. كل مشروع هنا فيه خطوات تمشي عليها، وحل مرجعي كامل شغّال (اتعمله compile واتشغّل واتأكد من مخرجاته). القاعدة: اكتبه بنفسك الأول في محرر على جهازك، وارجع للحل بس لما تخلص أو تقف تمامًا. ولما تخلص، علّم المشروع كمكتمل.

🧪
طريقة الشغل

اعمل فولدر projects/ على جهازك، وكل مشروع في ملف. اعمله compile بـ gcc -std=c17 -Wall -Wextra -g، وجرّبه على ملفات حقيقية (EXEs من C:\Windows\System32 مثلًا — للقراية بس). وبعد ما يشتغل: افتح الـ binary بتاعك في Ghidra وشوف كودك من الناحية التانية.

01

hexdump -C كامل

مبتدئ▼

أداة بتعرض أي ملف بالـ hex + ASCII، بالظبط زي hexdump -C وxxd. أول أداة لازم تبقى عندك.

freadprintf formatsisprintargv
  1. اقرا الملف من argv[1] بـ fopen(..., "rb") وchunks من 16 byte.
  2. اطبع السطر: offset بـ 8 خانات hex، 16 byte (مسافة زيادة بعد الـ 8)، وعمود ASCII بين |...|.
  3. السطر الأخير الناقص: املا الفراغ عشان عمود الـ ASCII يفضل في مكانه.
  4. ضغط التكرار: لو سطر مطابق للي قبله، اطبع * مرة واحدة وتخطّى (زي الأداة الحقيقية).
  5. Stretch: -s offset و-n length عشان تعرض جزء بس من ملف كبير.
#include #include #include static void dump_line(size_t off, const unsigned char *b, size_t n) { printf("%08zx ", off); for (size_t i = 0; i < 16; i++) { if (i < n) printf("%02x ", b[i]); else printf(" "); if (i == 7) putchar(' '); } printf(" |"); for (size_t i = 0; i < n; i++) putchar(isprint(b[i]) ? b[i] : '.'); printf("|\n"); } /* In the real tool, data comes from fread(); here it is embedded so it runs anywhere */ static void hexdump(const unsigned char *data, size_t len) { unsigned char prev[16]; int have_prev = 0, starred = 0; size_t off = 0; for (; off < len; off += 16) { size_t n = len - off < 16 ? len - off : 16; if (n == 16 && have_prev && memcmp(prev, data + off, 16) == 0) { if (!starred) { puts("*"); starred = 1; } continue; } dump_line(off, data + off, n); memcpy(prev, data + off, n); have_prev = (n == 16); starred = 0; } printf("%08zx\n", len); /* final offset = total size */ } int main(void) { static const char stub[] = "This program cannot be run in DOS mode.\r\n$"; unsigned char data[128] = { 0 }; /* zero lines repeat -> '*' */ memcpy(data, "MZ\x90\x00", 4); memcpy(data + 64, stub, sizeof stub - 1); /* 42 bytes: 64..105 */ hexdump(data, 64 + sizeof stub - 1); return 0; }

strings: ASCII + UTF-16LE

متوسط▼

نسخة من strings بتلاقي النوعين مع بعض (الأداة الأصلية بتحتاج -el للـ wide). أول خطوة triage في أي sample.

state machineUTF-16LEoffsets
  1. ASCII: تتابع bytes من 0x20 لـ 0x7E (و\t) طوله ≥ min (افتراضي 4).
  2. UTF-16LE: تتابع أزواج [printable, 0x00] طوله ≥ min حرف.
  3. اطبع لكل string: الـ offset بالـ hex، النوع (A/W)، والنص.
  4. Stretch: فلتر بـ regex بسيط (URLs، IPs، paths، registry keys) وعلّم المهم بـ [!].
#include #include #define MIN_LEN 4 static int is_text(unsigned char c) { return (c >= 0x20 && c <= 0x7E) || c == '\t'; } static void scan_ascii(const unsigned char *d, size_t n) { size_t start = 0, run = 0; for (size_t i = 0; i <= n; i++) { if (i < n && is_text(d[i])) { if (!run) start = i; run++; continue; } if (run >= MIN_LEN) printf("%06zx A %.*s\n", start, (int)run, (const char *)d + start); run = 0; } } static void scan_wide(const unsigned char *d, size_t n) { size_t start = 0, run = 0; /* run counts characters */ for (size_t i = 0; i + 1 <= n; ) { if (i + 1 < n && is_text(d[i]) && d[i + 1] == 0x00) { if (!run) start = i; run++; i += 2; continue; } if (run >= MIN_LEN) { printf("%06zx W ", start); for (size_t k = 0; k < run; k++) putchar(d[start + 2 * k]); putchar('\n'); } run = 0; i += 1; if (i >= n) break; } } int main(void) { const unsigned char blob[] = "\x00\x01" "GetProcAddress" "\x00\xFF" "h\0t\0t\0p\0:\0/\0/\0c\0" "2\0.\0t\0e\0s\0t\0" "\x00\x00" "\x90\x90" "abc" "\x01\x02" /* 3 chars: below the minimum */ "S\0o\0f\0t\0w\0a\0r\0e\0" "\x00\x00" "kernel32.dll"; size_t n = sizeof blob - 1; scan_ascii(blob, n); scan_wide(blob, n); return 0; }
03

Base64 decoder (+ custom alphabet)

متوسط▼

Base64 في كل حتة في الـ malware: configs، PowerShell payloads، بيانات الـ C2. وكتير من العائلات بتغيّر الـ alphabet عشان الأدوات الجاهزة متفكهاش — فلازم decoder بيقبل alphabet مخصص.

bit packinglookup tablevalidation
  1. ابنِ reverse table من 256 خانة من الـ alphabet (قيمة كل حرف 0..63، والباقي -1).
  2. كل 4 حروف = 24 bit = 3 bytes. اتعامل مع الـ = padding.
  3. ارفض أي حرف مش في الـ alphabet (ارجع error بدل ما تطلع زبالة).
  4. Stretch: لو الـ alphabet مخصص، جرّب تستنتجه من known plaintext.
#include #include #include static const char STD_ALPHA[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"; /* Returns decoded length, or -1 on invalid input */ static long b64_decode(const char *in, const char *alphabet, uint8_t *out, size_t out_cap) { int8_t rev[256]; memset(rev, -1, sizeof rev); for (int i = 0; i < 64; i++) rev[(unsigned char)alphabet[i]] = (int8_t)i; uint32_t acc = 0; int bits = 0; size_t n = 0; for (const char *p = in; *p && *p != '='; p++) { int v = rev[(unsigned char)*p]; if (v < 0) return -1; acc = (acc << 6) | (uint32_t)v; bits += 6; if (bits >= 8) { bits -= 8; if (n >= out_cap) return -1; out[n++] = (uint8_t)(acc >> bits); } } return (long)n; } int main(void) { uint8_t out[128]; long n = b64_decode("aHR0cDovL2V4YW1wbGUudGVzdC9nYXRlLnBocA==", STD_ALPHA, out, sizeof out); printf("standard : %.*s\n", (int)n, (char *)out); /* Same data, but the malware rotated its alphabet by 13 positions */ char custom[65]; for (int i = 0; i < 64; i++) custom[i] = STD_ALPHA[(i + 13) % 64]; custom[64] = '\0'; n = b64_decode("nUeBpQ18YDiFljC9oTh7qTiAqPK0lkeyY0O1pN==", custom, out, sizeof out); printf("custom : %.*s\n", (int)n, (char *)out); printf("invalid : %ld\n", b64_decode("abc$def", STD_ALPHA, out, sizeof out)); return 0; }

RC4 — الأشهر في الـ malware

متوسط▼

RC4 بسيط جدًا (حوالي 20 سطر) وعشان كده منتشر جدًا في تشفير الـ configs والـ traffic. لازم تعرفه من شكله في الـ Assembly: loop بيملا array من 256 بـ 0..255، وبعدين loop تاني بيعمل swap مع % keylen.

KSAPRGAtest vectors
  1. KSA: S[i] = i، وبعدين j = (j + S[i] + key[i % keylen]) % 256 وswap(S[i], S[j]).
  2. PRGA: لكل byte: i++، j += S[i]، swap، وout = in ^ S[(S[i] + S[j]) % 256].
  3. اتأكد من الـ test vectors الرسمية قبل ما تستخدمه على sample.
  4. Stretch: اكتبه بحيث تقدر تتعرف عليه في Ghidra: compile بـ -O2 وشوف الـ loops.
#include #include #include typedef struct { uint8_t S[256]; uint8_t i, j; } RC4; static void rc4_init(RC4 *c, const uint8_t *key, size_t keylen) { for (int k = 0; k < 256; k++) c->S[k] = (uint8_t)k; /* recognisable: 0..255 fill */ uint8_t j = 0; for (int k = 0; k < 256; k++) { j = (uint8_t)(j + c->S[k] + key[k % keylen]); /* recognisable: % keylen */ uint8_t t = c->S[k]; c->S[k] = c->S[j]; c->S[j] = t; } c->i = c->j = 0; } static void rc4_crypt(RC4 *c, uint8_t *buf, size_t len) { for (size_t n = 0; n < len; n++) { c->i = (uint8_t)(c->i + 1); c->j = (uint8_t)(c->j + c->S[c->i]); uint8_t t = c->S[c->i]; c->S[c->i] = c->S[c->j]; c->S[c->j] = t; buf[n] ^= c->S[(uint8_t)(c->S[c->i] + c->S[c->j])]; } } static void test(const char *key, const char *plain) { uint8_t buf[64]; size_t n = strlen(plain); memcpy(buf, plain, n); RC4 c; rc4_init(&c, (const uint8_t *)key, strlen(key)); rc4_crypt(&c, buf, n); printf("%-7s %-15s -> ", key, plain); for (size_t i = 0; i < n; i++) printf("%02X", buf[i]); rc4_init(&c, (const uint8_t *)key, strlen(key)); /* same op decrypts */ rc4_crypt(&c, buf, n); printf(" | back: %.*s\n", (int)n, (char *)buf); } int main(void) { test("Key", "Plaintext"); /* expected BBF316E8D940AF0AD3 */ test("Wiki", "pedia"); /* expected 1021BF0420 */ test("Secret", "Attack at dawn"); /* expected 45A01F645FC35B383552544B9BF5 */ return 0; }

XOR brute-forcer بالـ scoring

متوسط▼

لقيت blob مشفّر بـ XOR بـ byte واحد ومش عارف الـ key. جرّب الـ 256 احتمال، وادّي كل نتيجة score (نسبة الحروف المقروءة + وجود كلمات زي http و.exe)، واطبع أفضل نتايج. ده اللي بتعمله أدوات زي XORSearch وFLOSS.

brute forceheuristicsknown plaintext
  1. لكل key من 1 لـ 255: فك نسخة من الـ buffer.
  2. Score = عدد الـ bytes المقروءة + bonus كبير لو لقيت http أو .exe أو MZ في الأول.
  3. رتّب واطبع أعلى 3 keys بالـ score والنص.
  4. Stretch: multi-byte XOR key (طول 2..8): اكسره بـ known plaintext أو بتحليل التكرار.
#include #include #include #define LEN_MAX 128 static int score(const uint8_t *s, size_t n) { int sc = 0; for (size_t i = 0; i < n; i++) if ((s[i] >= 0x20 && s[i] <= 0x7E)) sc++; char tmp[LEN_MAX + 1]; memcpy(tmp, s, n); tmp[n] = '\0'; if (strstr(tmp, "http")) sc += 50; if (strstr(tmp, ".exe") || strstr(tmp, ".php")) sc += 30; return sc; } int main(void) { /* "http://198.51.100.23:8080/panel/gate.php" XOR'd with an unknown key */ const char *plain = "http://198.51.100.23:8080/panel/gate.php"; uint8_t enc[LEN_MAX]; size_t n = strlen(plain); for (size_t i = 0; i < n; i++) enc[i] = (uint8_t)plain[i] ^ 0x4B; /* the "unknown" key */ struct { int key, score; } best[3] = { { 0, -1 }, { 0, -1 }, { 0, -1 } }; uint8_t buf[LEN_MAX]; for (int k = 1; k < 256; k++) { for (size_t i = 0; i < n; i++) buf[i] = enc[i] ^ (uint8_t)k; int sc = score(buf, n); for (int r = 0; r < 3; r++) { /* keep a sorted top-3 */ if (sc > best[r].score) { for (int m = 2; m > r; m--) best[m] = best[m - 1]; best[r].key = k; best[r].score = sc; break; } } } for (int r = 0; r < 3; r++) { for (size_t i = 0; i < n; i++) buf[i] = enc[i] ^ (uint8_t)best[r].key; printf("#%d key=0x%02X score=%3d %.*s\n", r + 1, best[r].key, best[r].score, (int)n, (char *)buf); } return 0; }

PE Parser على header حقيقي

متقدم▼

الـ bytes اللي في الكود ده هي أول 792 byte من EXE حقيقي (برنامج hello world اتعمله compile بـ MinGW gcc 15.2 لـ Windows x64). الـ parser بيقراها بأمان ويطلع كل المعلومات اللي PE-bear بيعرضها. على جهازك: بدّل الـ array بـ fread لأي ملف.

bounds-checked readsheaderssectionsdata directories
  1. اتأكد من MZ، اقرا e_lfanew واتأكد إنه جوه الملف، وبعدين PE\0\0.
  2. اطبع الـ FILE_HEADER: Machine، عدد الـ sections، الـ timestamp، الـ characteristics.
  3. اطبع الـ OPTIONAL_HEADER64: Magic، EntryPoint، ImageBase، Subsystem، DllCharacteristics (فكّها لـ flags).
  4. اطبع جدول الـ sections مع صلاحيات RWX، وحدد الـ section اللي فيها الـ entry point.
  5. اطبع الـ data directories غير الفاضية.
  6. Stretch (على ملف كامل): امشي على الـ Import Table: لكل DLL اطبع اسمها ودوالها (RVA ← offset بالدالة اللي كتبناها في المرحلة 16).

اكتب Crackme واكسره بنفسك

متقدم▼

أحسن طريقة تتعلم بيها الـ RE: تكتب برنامج بيتحقق من serial، تعمله compile بـ -O2 -s، تنسى الكود، وتفتحه في Ghidra/x64dbg وتحاول تكسره بـ 3 طرق: patch، استخراج الـ serial وقت التشغيل، وkeygen.

hashingkeygenpatchingx64dbg
  1. اكتب int check(const char *name, const char *serial): الـ serial = hash للـ name متنسّق hex.
  2. اعمله compile مرتين: -O0 -g و-O2 -s. افتح الاتنين في Ghidra وقارن.
  3. Patch: لاقي الـ jz/jnz بعد الـ check واقلبه (في x64dbg أو hex editor).
  4. Runtime: حط breakpoint على strcmp وشوف الـ serial الصح في الـ registers.
  5. Keygen: أعد كتابة الـ hash من الـ disassembly واعمل keygen يطلع serial لأي اسم — ده الأصعب والأقيم.
#include #include #include /* The "secret" algorithm a reverser must recover */ static uint32_t name_hash(const char *name) { uint32_t h = 0x1505; for (const unsigned char *p = (const unsigned char *)name; *p; p++) { h = (h << 5) + h + *p; /* djb2-style */ h ^= 0xA5A5A5A5u; h = (h << 3) | (h >> 29); /* rol 3 */ } return h; } static int check(const char *name, const char *serial) { char expected[16]; if (strlen(name) < 4) return 0; snprintf(expected, sizeof expected, "CM-%08X", name_hash(name)); return strcmp(expected, serial) == 0; /* breakpoint here reveals the answer */ } /* What you write after reversing check() */ static void keygen(const char *name, char *out, size_t n) { snprintf(out, n, "CM-%08X", name_hash(name)); } int main(void) { const char *names[] = { "reverser", "analyst", "abc" }; char serial[16]; for (int i = 0; i < 3; i++) { keygen(names[i], serial, sizeof serial); printf("%-9s -> %s : %s\n", names[i], serial, check(names[i], serial) ? Bytecode VM + Disassembler

متقدم▼

أقوى أنواع الـ obfuscation (VMProtect، Themida، وكتير من الـ malware المخصص) بيحوّل الكود لـ bytecode لـ CPU وهمي، ومعاه interpreter. عشان تكسر ده لازم تفهم إزاي الـ VM بيتبني: fetch ← decode ← execute في loop مع switch (أو جدول handlers). هنا هتبني VM صغير وdisassembler ليه.

interpreter loopstack machinehandlersdisassembler
  1. عرّف الـ ISA: opcodes بـ byte واحد + operands (PUSH imm8، XOR، ADD، EMIT، DEC، JNZ rel8، HALT…).
  2. اكتب الـ disassembler الأول: بيطبع كل instruction بعنوانها — ده أول حاجة بتعملها مع VM حقيقي.
  3. اكتب الـ interpreter: pc، stack، وloop فيه switch (opcode). اتشيك على الـ stack overflow/underflow والـ pc.
  4. اكتب برنامج bytecode بيفك string مشفّر ويطبعه.
  5. Stretch: اعمل compile بـ -O2 وشوف الـ dispatcher في Ghidra: jump table كبير — ده شكل "VM handler table" اللي هتقابله.
