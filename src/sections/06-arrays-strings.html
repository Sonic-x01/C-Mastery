الـ Arrays والـ Strings: بيانات متتالية في الذاكرة

الـ Array = عناصر من نفس النوع جنب بعض في الذاكرة بدون أي فراغ. والـ string في C مش نوع خاص — هو مجرد array من char بيخلص بـ byte قيمته صفر ('\0'). مفيش طول متخزن، ومفيش حماية من إنك تكتب برا الحدود. وده بالظبط سبب أشهر ثغرات التاريخ.

الـ Arrays: التعريف والوصول arrays

#include int main(void) { int ports[5] = { 80, 443, 8080 }; /* missing elements become 0 */ int ids[] = { 7, 3, 9, 1 }; /* size deduced: 4 */ int zeros[100] = { 0 }; /* all zero */ size_t n = sizeof ids / sizeof ids[0]; /* element count idiom */ printf("ids has %zu elements, %zu bytes\n", n, sizeof ids); for (size_t i = 0; i < 5; i++) printf("ports[%zu] = %d\n", i, ports[i]); ports[3] = 4444; /* write */ printf("ports[3] now = %d, zeros[99] = %d\n", ports[3], zeros[99]); /* a[i] is EXACTLY *(a + i): even i[a] compiles (don't do it!) */ printf("ids[2] = %d, *(ids + 2) = %d, 2[ids] = %d\n", ids[2], *(ids + 2), 2[ids]); return 0; }

الـ Array في الـ Assembly

الوصول لعنصر = base + index × size. هتشوفه كده: mov eax, DWORD PTR [rbp-32+rax*4] (array من int على الـ stack) أو movzx eax, BYTE PTR [rdx+rax] (array من bytes). الـ scale (×1، ×2، ×4، ×8) بيقولك حجم العنصر = نوعه.

C مش بتتحقق من الحدود

ports[10] = 1; بيعدّي الـ compile عادي وبيكتب في ذاكرة مش بتاعتك = Undefined Behavior. ممكن يقع، ممكن يبوظ متغير تاني بصمت، وممكن (لو المهاجم متحكم في الـ index أو البيانات) يتحول لـ code execution.

Arrays متعددة الأبعاد row-major

#include int main(void) { unsigned char sbox[3][4] = { { 0x63, 0x7C, 0x77, 0x7B }, { 0xF2, 0x6B, 0x6F, 0xC5 }, { 0x30, 0x01, 0x67, 0x2B }, }; for (int r = 0; r < 3; r++) { for (int c = 0; c < 4; c++) printf("%02X ", sbox[r][c]); printf("\n"); } /* Row-major: [r][c] lives at offset r*COLS + c in one flat block */ unsigned char *flat = &sbox[0][0]; printf("sbox[1][2] = 0x%02X = flat[1*4+2] = 0x%02X\n", sbox[1][2], flat[1 * 4 + 2]); return 0; }
🔬
S-Box = بصمة خوارزمية تشفير

الـ bytes 63 7C 77 7B F2 6B 6F C5 هي أول الـ AES S-Box. لما تلاقي جدول ثابت كبير في .rdata، دوّر على أول bytes منه في جوجل — غالبًا هيطلع ثابت لخوارزمية معروفة (AES، CRC32، MD5، SHA-256، RC4…). أدوات زي FindCrypt (IDA/Ghidra) وcapa بتعمل ده أوتوماتيك.

الدوال دي غالبًا مش هتلاقيها كـ calls

في -O2، gcc وMSVC بيعملوا inline لـ strlen وmemcpy وmemset بأحجام صغيرة: memcpy لـ 16 byte بتبقى movdqu + movups، وmemset بيبقى سلسلة mov qword ptr. ولما تلاقي loop بتعد لحد ما تلاقي byte = 0 → ده strlen مكتوبة يدوي (الـ malware بيعمل كده كتير عشان يقلل الـ imports).

الـ Arrays بتتحول لـ Pointers decay

لما تبعت array لدالة، اللي بيتبعت فعليًا هو pointer لأول عنصر — والحجم بيضيع. عشان كده لازم تبعت الطول معاه دايمًا.

#include static void show(int arr[], size_t n) { /* int arr[] == int *arr here */ printf("inside : sizeof(arr) = %zu (a pointer!)\n", sizeof(arr)); for (size_t i = 0; i < n; i++) printf("%d ", arr[i]); printf("\n"); } int main(void) { int data[6] = { 4, 8, 15, 16, 23, 42 }; printf("outside: sizeof(data) = %zu\n", sizeof data); show(data, sizeof data / sizeof data[0]); return 0; }

الـ compiler نفسه بيحذرك (-Wsizeof-array-argument) — شغّل وشوف.

Wide Strings: لغة Windows wchar_t · UTF-16

Windows بيستخدم UTF-16LE داخليًا: كل حرف إنجليزي = 2 bytes ('A' = 41 00). كل API ليها نسختين: CreateFileA (ANSI) وCreateFileW (Wide). على Linux wchar_t = 4 bytes.

#include #include #include int main(void) { const wchar_t *w = L"cmd"; printf("sizeof(wchar_t) here = %zu (Windows: 2)\n", sizeof(wchar_t)); printf("wcslen = %zu chars\n", wcslen(w)); /* What "cmd" looks like as UTF-16LE in a Windows binary */ const unsigned char utf16le[] = { 'c', 0, 'm', 0, 'd', 0, 0, 0 }; printf("UTF-16LE bytes: "); for (size_t i = 0; i < sizeof utf16le; i++) printf("%02X ", utf16le[i]); printf("\n"); /* Convert UTF-16LE (ASCII range) back to a normal string */ char out[8] = { 0 }; for (size_t i = 0; utf16le[i * 2] != 0; i++) out[i] = (char)utf16le[i * 2]; printf("decoded: %s (len %zu)\n", out, strlen(out)); return 0; }
